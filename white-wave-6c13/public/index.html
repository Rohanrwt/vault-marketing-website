export default { async fetch(request) { const url = new URL(request.url); let filePath = url.pathname; // If root, serve index.html if (filePath === '/' || filePath === '') { filePath = '/index.html'; } // If no extension, add .html if (!filePath.includes('.')) { filePath += '.html'; } // Get file from your GitHub repo const githubUrl = `https://raw.githubusercontent.com/Rohanrwt/vault-marketing-website/main${filePath}`; try { const response = await fetch(githubUrl); if (response.status === 404) { return new Response('404 Not Found', { status: 404 }); } return new Response(response.body, { status: response.status, headers: response.headers }); } catch (error) { return new Response('Error loading file', { status: 500 }); } } };
